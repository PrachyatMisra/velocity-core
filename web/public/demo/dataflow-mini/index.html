Transform (Pandas)

df = pd.DataFrame(raw)
df["country"] = df["country"].str.lower()
df["amount"]  = df["amount"].astype(float)
df["active"]  = df["active"].map({"true":True,"false":False})
# validation: required field "name" must be non-empty
clean = df[df["name"].str.len() > 0].copy()
rejected = df[df["name"].str.len() == 0]
clean.to_sql("users", engine, if_exists="append", index=False)
